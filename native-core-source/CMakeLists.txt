cmake_minimum_required(VERSION 3.22)
project(homevpn_official_native LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/SoftEtherVPN_Stable")
set(OPENSSL_SOURCE_ROOT "" CACHE PATH "Pinned OpenSSL source directory")
set(OPENSSL_BUILD_ROOT "" CACHE PATH "OpenSSL build directory for the selected Android ABI")
if(NOT ANDROID_ABI OR NOT OPENSSL_SOURCE_ROOT OR NOT OPENSSL_BUILD_ROOT)
  message(FATAL_ERROR "Set ANDROID_ABI, OPENSSL_SOURCE_ROOT and OPENSSL_BUILD_ROOT")
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(SE_WORD_BITS 64)
else()
  set(SE_WORD_BITS 32)
endif()
file(STRINGS "${SE_ROOT}/src/makefiles/linux_${SE_WORD_BITS}bit.mak" OBJECT_LINES REGEX "^OBJECTS_(MAYAQUA|CEDAR)=")
set(SE_SOURCES "")
foreach(LINE IN LISTS OBJECT_LINES)
  string(REGEX REPLACE "^[^=]+=" "" LINE "${LINE}")
  separate_arguments(OBJECTS UNIX_COMMAND "${LINE}")
  foreach(OBJECT IN LISTS OBJECTS)
    string(REPLACE "tmp/objs/" "${SE_ROOT}/src/" SOURCE "${OBJECT}")
    string(REGEX REPLACE "\\.o$" ".c" SOURCE "${SOURCE}")
    list(APPEND SE_SOURCES "${SOURCE}")
  endforeach()
endforeach()
add_library(official_core STATIC ${SE_SOURCES} compat/android_console.c compat/android_runtime.c compat/android_frame.c)
target_include_directories(official_core PUBLIC
  "${SE_ROOT}/src" "${SE_ROOT}/src/Mayaqua" "${SE_ROOT}/src/Cedar"
  "${CMAKE_CURRENT_SOURCE_DIR}/compat/include" "${OPENSSL_BUILD_ROOT}/include" "${OPENSSL_SOURCE_ROOT}/include")
target_compile_definitions(official_core PUBLIC UNIX UNIX_LINUX NDEBUG VPN_SPEED)
if(SE_WORD_BITS EQUAL 64)
  target_compile_definitions(official_core PUBLIC CPU_64)
else()
  target_compile_definitions(official_core PUBLIC _FILE_OFFSET_BITS=64)
endif()
target_compile_options(official_core PUBLIC -fsigned-char -fstack-protector-strong
  "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=native-core-source"
  -Wno-deprecated-declarations -Wno-pointer-sign -Wno-void-pointer-to-int-cast)
target_link_libraries(official_core PUBLIC "${OPENSSL_BUILD_ROOT}/libssl.a" "${OPENSSL_BUILD_ROOT}/libcrypto.a" z dl m log android)
foreach(PROGRAM vpnclient vpncmd)
  if(PROGRAM STREQUAL "vpnclient")
    set(ENTRY "${SE_ROOT}/src/vpnclient/vpncsvc.c")
  else()
    set(ENTRY "${SE_ROOT}/src/vpncmd/vpncmd.c")
  endif()
  add_executable(${PROGRAM} "${ENTRY}")
  target_link_libraries(${PROGRAM} PRIVATE official_core)
  target_link_options(${PROGRAM} PRIVATE -pie -Wl,-z,relro,-z,now,-z,noexecstack -Wl,-z,max-page-size=16384 -Wl,--build-id=sha1)
  set_target_properties(${PROGRAM} PROPERTIES OUTPUT_NAME "libhomevpn_${PROGRAM}" SUFFIX ".so"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/out/${ANDROID_ABI}")
endforeach()
